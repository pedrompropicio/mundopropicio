-- ============== DRY-RUN (so leitura) ==========================
WITH ev AS (
  SELECT id FROM events WHERE name = 'Coala Festival Portugal 2026'
),
cats AS (
  SELECT id FROM account_categories WHERE code IN ('1.2.01','1.2.02')
),
bp AS (
  SELECT f.id, f.amount FROM event_forecasts f
  JOIN ev ON ev.id = f.event_id
  WHERE f.category_id IN (SELECT id FROM cats)
),
tx AS (
  SELECT t.id, t.amount FROM transactions t
  JOIN ev ON ev.id = t.event_id
  WHERE t.category_id IN (SELECT id FROM cats)
)
SELECT
  (SELECT COUNT(*) FROM bp) AS bp_linhas,
  (SELECT COALESCE(SUM(amount),0) FROM bp) AS bp_total_eur,
  (SELECT COUNT(*) FROM tx) AS tx_linhas,
  (SELECT COALESCE(SUM(amount),0) FROM tx) AS tx_total_eur;
