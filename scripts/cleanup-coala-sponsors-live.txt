-- ============== DRY-RUN (so leitura) ==========================
WITH ev AS (
  SELECT id, name FROM events WHERE name = 'Coala Festival Portugal 2026'
),
cats AS (
  SELECT id, code, name FROM account_categories WHERE code IN ('1.2.01','1.2.02')
)
SELECT 'EVENTO' AS bloco, ev.id::text AS id, ev.name AS desc, NULL::numeric AS valor, NULL AS status
FROM ev
UNION ALL
SELECT 'BP', f.id::text, COALESCE(f.description,'(sem descricao)') || ' [' || c.code || ']',
       f.amount, f.status::text
FROM event_forecasts f
JOIN ev ON ev.id = f.event_id
JOIN cats c ON c.id = f.category_id
UNION ALL
SELECT 'TX', t.id::text, COALESCE(t.description,'(sem descricao)') || ' [' || c.code || ']',
       t.amount, t.status::text
FROM transactions t
JOIN ev ON ev.id = t.event_id
JOIN cats c ON c.id = t.category_id
ORDER BY bloco, desc;

-- Totais agregados
WITH ev AS (SELECT id FROM events WHERE name = 'Coala Festival Portugal 2026'),
cats AS (SELECT id FROM account_categories WHERE code IN ('1.2.01','1.2.02'))
SELECT
  (SELECT COUNT(*) FROM event_forecasts f JOIN ev ON ev.id=f.event_id WHERE f.category_id IN (SELECT id FROM cats)) AS bp_linhas,
  (SELECT COALESCE(SUM(amount),0) FROM event_forecasts f JOIN ev ON ev.id=f.event_id WHERE f.category_id IN (SELECT id FROM cats)) AS bp_total,
  (SELECT COUNT(*) FROM transactions t JOIN ev ON ev.id=t.event_id WHERE t.category_id IN (SELECT id FROM cats)) AS tx_linhas,
  (SELECT COALESCE(SUM(amount),0) FROM transactions t JOIN ev ON ev.id=t.event_id WHERE t.category_id IN (SELECT id FROM cats)) AS tx_total;
