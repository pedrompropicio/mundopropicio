-- ╔══════════════════════════════════════════════════════════════════════════╗
-- ║ BLOCO 1 — DRY-RUN (só consulta, não apaga nada)                          ║
-- ╚══════════════════════════════════════════════════════════════════════════╝

WITH ev AS (
  SELECT id FROM events WHERE name ILIKE '%Coala%Portugal%2026%'
),
cats AS (
  SELECT id FROM account_categories WHERE code IN ('1.2.01','1.2.02')
)
SELECT
  (SELECT count(*) FROM event_forecasts f
     WHERE f.event_id IN (SELECT id FROM ev)
       AND f.category_id IN (SELECT id FROM cats))                AS bp_linhas,
  (SELECT COALESCE(sum(amount),0) FROM event_forecasts f
     WHERE f.event_id IN (SELECT id FROM ev)
       AND f.category_id IN (SELECT id FROM cats))                AS bp_total_eur,
  (SELECT count(*) FROM transactions t
     WHERE t.event_id IN (SELECT id FROM ev)
       AND t.category_id IN (SELECT id FROM cats))                AS tx_linhas,
  (SELECT COALESCE(sum(amount),0) FROM transactions t
     WHERE t.event_id IN (SELECT id FROM ev)
       AND t.category_id IN (SELECT id FROM cats))                AS tx_total_eur;


-- ╔══════════════════════════════════════════════════════════════════════════╗
-- ║ BLOCO 2 — DELETE DEFINITIVO (corre tudo de uma vez)                      ║
-- ║ Devolve uma linha NOTICE com os contadores reais de cada DELETE.         ║
-- ╚══════════════════════════════════════════════════════════════════════════╝

DO $$
DECLARE
  v_event_ids uuid[];
  v_cat_ids   uuid[];
  v_tx_ids    uuid[];
  v_fc_ids    uuid[];
  n_link int := 0; n_pay int := 0; n_doc int := 0; n_tx int := 0; n_bp int := 0;
BEGIN
  SELECT array_agg(id) INTO v_event_ids
    FROM events WHERE name ILIKE '%Coala%Portugal%2026%';
  SELECT array_agg(id) INTO v_cat_ids
    FROM account_categories WHERE code IN ('1.2.01','1.2.02');

  SELECT array_agg(id) INTO v_tx_ids FROM transactions
    WHERE event_id = ANY(v_event_ids) AND category_id = ANY(v_cat_ids);
  SELECT array_agg(id) INTO v_fc_ids FROM event_forecasts
    WHERE event_id = ANY(v_event_ids) AND category_id = ANY(v_cat_ids);

  RAISE NOTICE 'ALVO: events=%, cats=%, tx=%, fc=%',
    array_length(v_event_ids,1), array_length(v_cat_ids,1),
    COALESCE(array_length(v_tx_ids,1),0), COALESCE(array_length(v_fc_ids,1),0);

  -- 1) desligar Pipeline
  UPDATE sponsorship_pipeline
     SET linked_transaction_id = NULL, linked_forecast_id = NULL
   WHERE linked_transaction_id = ANY(COALESCE(v_tx_ids, ARRAY[]::uuid[]))
      OR linked_forecast_id    = ANY(COALESCE(v_fc_ids, ARRAY[]::uuid[]));
  GET DIAGNOSTICS n_link = ROW_COUNT;

  -- 2) pagamentos parcelados
  DELETE FROM transaction_payments
   WHERE transaction_id = ANY(COALESCE(v_tx_ids, ARRAY[]::uuid[]));
  GET DIAGNOSTICS n_pay = ROW_COUNT;

  -- 3) documentos anexados
  DELETE FROM transaction_documents
   WHERE transaction_id = ANY(COALESCE(v_tx_ids, ARRAY[]::uuid[]));
  GET DIAGNOSTICS n_doc = ROW_COUNT;

  -- 4) transações
  DELETE FROM transactions
   WHERE id = ANY(COALESCE(v_tx_ids, ARRAY[]::uuid[]));
  GET DIAGNOSTICS n_tx = ROW_COUNT;

  -- 5) linhas BP
  DELETE FROM event_forecasts
   WHERE id = ANY(COALESCE(v_fc_ids, ARRAY[]::uuid[]));
  GET DIAGNOSTICS n_bp = ROW_COUNT;

  RAISE NOTICE 'APAGADO: pipeline_unlinked=%, payments=%, docs=%, tx=%, bp=%',
    n_link, n_pay, n_doc, n_tx, n_bp;
END $$;


-- ╔══════════════════════════════════════════════════════════════════════════╗
-- ║ BLOCO 3 — VERIFICAÇÃO (deve devolver tudo a zero)                        ║
-- ╚══════════════════════════════════════════════════════════════════════════╝

WITH ev AS (
  SELECT id FROM events WHERE name ILIKE '%Coala%Portugal%2026%'
),
cats AS (
  SELECT id FROM account_categories WHERE code IN ('1.2.01','1.2.02')
)
SELECT
  (SELECT count(*) FROM event_forecasts f
     WHERE f.event_id IN (SELECT id FROM ev)
       AND f.category_id IN (SELECT id FROM cats))                AS bp_restantes,
  (SELECT count(*) FROM transactions t
     WHERE t.event_id IN (SELECT id FROM ev)
       AND t.category_id IN (SELECT id FROM cats))                AS tx_restantes;
