-- ============================================================
-- FLATTEN 10.7.* L4 -> L3  (LIVE v5, multi-tenant focado)
-- Para empresas 7c858982... e 7d831e59...
-- Estado actual:
--   10.7 (L2 "Serviços e Estrutura")
--     10.7.01 Serviços   -> contém 3 folhas L4 (Contab, Jurídico, Consultoria)
--     10.7.02 Estrutura  -> contém 3 folhas L4 (Aluguer, Energia, Internet)
--     10.7.03 Tecnologia -> contém 3 folhas L4 (Softwares, Cloud, Equipamentos)
-- Estado alvo:
--   10.7 Serviços   (L2) -> 10.7.01..03 (L3)
--   10.8 Estrutura  (L2) -> 10.8.01..03 (L3)
--   10.9 Tecnologia (L2) -> 10.9.01..03 (L3)
-- IDs das folhas L4 são preservados (BP/TX continuam vinculados).
-- Os 3 pais L3 antigos (10.7.01/02/03) são apagados no fim.
-- ============================================================

BEGIN;

-- 0) Empresas alvo
DROP TABLE IF EXISTS _co;
CREATE TEMP TABLE _co AS
SELECT unnest(ARRAY[
  '7c858982-6ccd-47ca-bd65-e0dd3eebf01c'::uuid,
  '7d831e59-6e82-427b-95a0-64904aae5dd2'::uuid
]) AS company_id;

-- 1) Renomear 10.7 -> "Serviços" (mantém o ID; passa a ser o pai dos serviços)
UPDATE account_categories ac
SET name = 'Serviços'
FROM _co
WHERE ac.company_id = _co.company_id
  AND ac.code = '10.7';

-- 2) Criar 10.8 Estrutura e 10.9 Tecnologia (L2, filhos de "10") por empresa, se não existirem
INSERT INTO account_categories (id, company_id, parent_id, code, name)
SELECT gen_random_uuid(), _co.company_id, p10.id, '10.8', 'Estrutura'
FROM _co
JOIN account_categories p10
  ON p10.company_id = _co.company_id AND p10.code = '10' AND p10.parent_id IS NULL
WHERE NOT EXISTS (
  SELECT 1 FROM account_categories x
  WHERE x.company_id = _co.company_id AND x.code = '10.8'
);

INSERT INTO account_categories (id, company_id, parent_id, code, name)
SELECT gen_random_uuid(), _co.company_id, p10.id, '10.9', 'Tecnologia'
FROM _co
JOIN account_categories p10
  ON p10.company_id = _co.company_id AND p10.code = '10' AND p10.parent_id IS NULL
WHERE NOT EXISTS (
  SELECT 1 FROM account_categories x
  WHERE x.company_id = _co.company_id AND x.code = '10.9'
);

-- 3) Re-parentar + renumerar as 17 folhas L4 (preservando IDs)
-- 3a) Folhas de 10.7.01.* -> filhos directos de 10.7 com códigos 10.7.01..03
WITH p AS (
  SELECT ac.id AS new_parent, ac.company_id
  FROM account_categories ac
  WHERE ac.code = '10.7'
    AND ac.company_id IN (SELECT company_id FROM _co)
)
UPDATE account_categories l
SET parent_id = p.new_parent,
    code = CASE l.code
      WHEN '10.7.01.01' THEN '10.7.01'
      WHEN '10.7.01.02' THEN '10.7.01' -- placeholder, corrigido abaixo
      ELSE l.code
    END
FROM p
WHERE l.company_id = p.company_id
  AND l.code LIKE '10.7.01.%';

-- (corrigir os códigos 10.7.01.* individualmente p/ não chocar com unique)
UPDATE account_categories SET code = '10.7.02' WHERE code = '10.7.01.02' AND company_id IN (SELECT company_id FROM _co);
UPDATE account_categories SET code = '10.7.03' WHERE code = '10.7.01.03' AND company_id IN (SELECT company_id FROM _co);

-- 3b) Folhas de 10.7.02.* -> filhos de 10.8 com códigos 10.8.01..03
WITH p AS (
  SELECT ac.id AS new_parent, ac.company_id
  FROM account_categories ac
  WHERE ac.code = '10.8'
    AND ac.company_id IN (SELECT company_id FROM _co)
)
UPDATE account_categories l
SET parent_id = p.new_parent,
    code = CASE l.code
      WHEN '10.7.02.01' THEN '10.8.01'
      WHEN '10.7.02.02' THEN '10.8.02'
      WHEN '10.7.02.03' THEN '10.8.03'
      ELSE l.code
    END
FROM p
WHERE l.company_id = p.company_id
  AND l.code LIKE '10.7.02.%';

-- 3c) Folhas de 10.7.03.* -> filhos de 10.9 com códigos 10.9.01..03
WITH p AS (
  SELECT ac.id AS new_parent, ac.company_id
  FROM account_categories ac
  WHERE ac.code = '10.9'
    AND ac.company_id IN (SELECT company_id FROM _co)
)
UPDATE account_categories l
SET parent_id = p.new_parent,
    code = CASE l.code
      WHEN '10.7.03.01' THEN '10.9.01'
      WHEN '10.7.03.02' THEN '10.9.02'
      WHEN '10.7.03.03' THEN '10.9.03'
      ELSE l.code
    END
FROM p
WHERE l.company_id = p.company_id
  AND l.code LIKE '10.7.03.%';

-- 4) Apagar os 3 pais L3 antigos (10.7.01/02/03) — agora sem filhos
DELETE FROM account_categories
WHERE company_id IN (SELECT company_id FROM _co)
  AND code IN ('10.7.01','10.7.02','10.7.03')
  AND id IN (
    'd1072000-0000-0000-0000-000000000001'::uuid,
    'd1073000-0000-0000-0000-000000000001'::uuid,
    'd1074000-0000-0000-0000-000000000001'::uuid,
    '7c754989-68bb-4ecc-9b2b-00651841e9ae'::uuid,
    '56226c2d-de68-4253-8adc-22cb7504f9cb'::uuid,
    'c97a8757-416d-4607-a59d-5fce3f9a3a7c'::uuid
  );

-- 5) AUDITORIA por empresa
WITH RECURSIVE tree AS (
  SELECT id, parent_id, code, company_id, 1 AS depth
  FROM account_categories WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id, c.parent_id, c.code, c.company_id, t.depth+1
  FROM account_categories c JOIN tree t ON c.parent_id = t.id
)
SELECT
  company_id,
  COUNT(*) FILTER (WHERE depth >= 4) AS l4_hierarquia,
  COUNT(*) FILTER (WHERE code ~ '^10\.7\.[0-9]+\.[0-9]+$') AS l4_codigo,
  COUNT(*) FILTER (WHERE code IN ('10.7.01','10.7.02','10.7.03')) AS antigos_grupos,
  COUNT(*) FILTER (WHERE code IN ('10.7','10.8','10.9')) AS novos_l2,
  COUNT(*) FILTER (WHERE code ~ '^10\.[789]\.[0-9]+$') AS novas_folhas_l3
FROM tree
WHERE company_id IN (SELECT company_id FROM _co)
GROUP BY company_id
ORDER BY company_id;

-- Esperado por empresa:
--   l4_hierarquia=0 | l4_codigo=0 | antigos_grupos=0
--   novos_l2=3 | novas_folhas_l3=9 (7c tem 8: faltava Softwares)
