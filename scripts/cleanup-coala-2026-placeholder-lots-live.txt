-- Remoção dos 8 lotes-placeholder (quantity=0, 0 vendas) que estavam a
-- inflar o Forecast do Simulador. Bug observado 2026-05-27: solver usava
-- preço destes lotes (900€/300€) como fallbackPrice, fazendo o ticket

  (SELECT COUNT(*) FROM ticket_sales s WHERE s.lot_id = l.id) AS sales_count
FROM event_ticket_lots l
WHERE l.id IN (

  'e3431010-d402-4fae-a987-7ba3fec1527f',
  '4712e2b7-6c18-48ae-81b2-40b976581594',
  '7f8ac550-aed3-4861-a318-14497e40d7fd',
  'd5030691-2d59-4bfa-b0ad-2e49042a4814'
);

DELETE FROM event_ticket_lots
WHERE id IN (
  '8fa503ae-b308-4ebc-a565-bf665bb1b9d6',
