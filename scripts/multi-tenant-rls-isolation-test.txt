-- ============================================================
-- BLOCO 7 — Cobertura: 66 tabelas isoladas em "varredura cega"
-- ============================================================
\echo ''
\echo '═══════════════════════════════════════════════════════════'
\echo ' BLOCO 7 — Varredura cega: para cada tabela com company_id, '
\echo ' D2 user nunca deve ver linhas com company_id <> Demo 2     '
\echo '═══════════════════════════════════════════════════════════'

BEGIN;
SET LOCAL ROLE authenticated;
SET LOCAL "request.jwt.claims" TO '{"sub":"00000000-0000-0000-0000-000000000d02","role":"authenticated"}';

DO $$
DECLARE
  r record;
  leaked bigint;
  total_leaked bigint := 0;
  total_tables int := 0;
BEGIN
  FOR r IN
    SELECT c.relname AS tablename
    FROM pg_class c
    JOIN pg_namespace n ON n.oid = c.relnamespace
    JOIN information_schema.columns col
      ON col.table_schema='public' AND col.table_name=c.relname AND col.column_name='company_id'
    WHERE n.nspname='public' AND c.relkind='r'
    ORDER BY c.relname
  LOOP
    total_tables := total_tables + 1;
    EXECUTE format(
      'SELECT count(*) FROM public.%I WHERE company_id <> %L::uuid',
      r.tablename,
      '6e174fca-69b6-4173-9aca-11a0a8355840'
    ) INTO leaked;
    IF leaked > 0 THEN
      RAISE WARNING 'LEAK em %: % linhas visíveis com company_id != Demo 2', r.tablename, leaked;
      total_leaked := total_leaked + leaked;
    END IF;
  END LOOP;
  RAISE NOTICE '── Varredura concluída: % tabelas testadas, % linhas leaked total', total_tables, total_leaked;
  IF total_leaked = 0 THEN
    RAISE NOTICE '✅ ISOLAMENTO RLS 100%% — 0 leakage cross-tenant';
  ELSE
    RAISE WARNING '❌ ISOLAMENTO COMPROMETIDO — % leaks no total', total_leaked;
  END IF;
END $$;

ROLLBACK;
