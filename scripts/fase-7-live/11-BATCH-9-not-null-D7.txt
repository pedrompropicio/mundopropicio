-- 1) Pré-check: tabelas que ainda têm linhas com company_id NULL
DO $$
DECLARE
  r record;
  v_total int;
  v_problemas text[] := '{}';
BEGIN
  FOR r IN
    SELECT table_name FROM information_schema.columns
    WHERE table_schema='public' AND column_name='company_id' AND is_nullable='YES'
    ORDER BY table_name
  LOOP
    EXECUTE format('SELECT count(*) FROM public.%I WHERE company_id IS NULL', r.table_name) INTO v_total;
    IF v_total > 0 THEN
      v_problemas := v_problemas || format('%s: %s linhas órfãs', r.table_name, v_total);
    END IF;
  END LOOP;
  IF array_length(v_problemas,1) IS NOT NULL THEN
    RAISE EXCEPTION 'NÃO PROSSEGUIR — corrigir órfãs primeiro: %', v_problemas;
  END IF;
  RAISE NOTICE '✅ Pré-check OK — todas as tabelas estão prontas para NOT NULL';
END $$;

-- 2) Aplicar NOT NULL
DO $$
DECLARE
  r record;
  v_count int := 0;
  v_failed text[] := '{}';
BEGIN
  FOR r IN
    SELECT table_name FROM information_schema.columns
    WHERE table_schema='public' AND column_name='company_id' AND is_nullable='YES'
      AND table_name NOT IN ('user_roles')  -- platform_admin tem company_id NULL legitimamente
    ORDER BY table_name
  LOOP
    BEGIN
      EXECUTE format('ALTER TABLE public.%I ALTER COLUMN company_id SET NOT NULL', r.table_name);
      v_count := v_count + 1;
    EXCEPTION WHEN OTHERS THEN
      v_failed := v_failed || format('%s: %s', r.table_name, SQLERRM);
    END;
  END LOOP;
  RAISE NOTICE '✅ NOT NULL aplicado em % tabelas', v_count;
  IF array_length(v_failed,1) IS NOT NULL THEN
    RAISE WARNING '❌ Falhas: %', v_failed;
  END IF;
END $$;

-- 3) Validação
