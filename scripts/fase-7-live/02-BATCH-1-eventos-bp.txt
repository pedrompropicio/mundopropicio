-- 4) Função helper para verificar se uma linha pertence à empresa do utilizador
-- (usa NULL-safe: se a linha tem company_id NULL, considera que pertence — fase de transição)
CREATE OR REPLACE FUNCTION public.row_belongs_to_current_company(_row_company_id uuid)
RETURNS boolean
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT
    public.is_platform_admin()
    OR _row_company_id IS NULL
    OR _row_company_id = public.current_company_id()
$$;

-- 5) Policies RESTRICTIVE de isolamento por empresa
-- Estas policies são APLICADAS EM CONJUNTO com as existentes (AND).
-- Não removem qualquer permissão; apenas filtram por empresa.

-- events
DROP POLICY IF EXISTS "company_isolation_events" ON public.events;
CREATE POLICY "company_isolation_events" ON public.events
  AS RESTRICTIVE FOR ALL
  TO authenticated
  USING (public.row_belongs_to_current_company(company_id))
  WITH CHECK (public.row_belongs_to_current_company(company_id));

-- 6) Trigger genérico para preencher company_id automaticamente em INSERT
-- (lê de current_company_id() do utilizador autenticado se a coluna ficar NULL)
CREATE OR REPLACE FUNCTION public.set_company_id_on_insert()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF NEW.company_id IS NULL THEN
    NEW.company_id := public.current_company_id();
  END IF;
  RETURN NEW;
END;
$$;

-- 7) Aplicar trigger BEFORE INSERT às tabelas
DO $$
DECLARE
  v_table text;
  v_tables text[] := ARRAY[
    'events','event_dates','event_sessions','event_partners','event_forecasts',
    'event_forecast_partners','event_implementations','event_closing_costs',
    'event_partner_extras','bp_versions','bp_version_audit_log',
    'bp_orphan_attachments','event_forecast_formalidade_log','forecast_audit_log'
  ];
BEGIN
  FOREACH v_table IN ARRAY v_tables LOOP
    EXECUTE format('DROP TRIGGER IF EXISTS trg_set_company_id ON public.%I', v_table);
    EXECUTE format(
      'CREATE TRIGGER trg_set_company_id BEFORE INSERT ON public.%I FOR EACH ROW EXECUTE FUNCTION public.set_company_id_on_insert()',
      v_table
    );
  END LOOP;
END $$;
