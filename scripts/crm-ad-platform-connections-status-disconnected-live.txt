-- Estende o CHECK constraint de status em crm.ad_platform_connections para
-- aceitar 'disconnected' (utilizador desativou manualmente em /audience/connections).
-- Idempotente: pode ser corrido várias vezes.
--
-- Aplicar em: Live (Supabase Dashboard > SQL Editor)
-- Data: 2026-05-12

ALTER TABLE crm.ad_platform_connections
  DROP CONSTRAINT IF EXISTS ad_platform_connections_status_check;

ALTER TABLE crm.ad_platform_connections
  ADD CONSTRAINT ad_platform_connections_status_check
  CHECK (status = ANY (ARRAY['active','expired','revoked','error','disconnected']));

-- Verificar:
SELECT pg_get_constraintdef(c.oid)
FROM pg_constraint c
JOIN pg_class t ON t.oid = c.conrelid
JOIN pg_namespace n ON n.oid = t.relnamespace
WHERE n.nspname = 'crm'
  AND t.relname = 'ad_platform_connections'
  AND c.conname = 'ad_platform_connections_status_check';
