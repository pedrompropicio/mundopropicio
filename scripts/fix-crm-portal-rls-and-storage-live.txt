-- Security fixes para tabelas/bucket criados directamente em Live (Portal MP + CRM).
-- Correr no SQL editor do Supabase dashboard (Live: ukpuhoynrqobqtzdbysp).
--
-- Cobre:
--   1) GRANTs + RLS company-scoped em contacts, leads
--   2) GRANTs + RLS company-scoped em event_faqs, event_lineups
--   3) Storage bucket crm-meta-creatives: isolamento por path (company-scoped)
--      via RESTRICTIVE policies usando storage_path_belongs_to_current_company.
--
-- Padrão idêntico ao usado nas outras tabelas multi-tenant:
--   - mutações: admin/manager (e service_role bypass implícito)
--   - SELECT: membros do tenant via current_company_id()
--   - INSERT a partir do portal público (anon) só em contacts/leads quando vier
--     via edge function (que usa SERVICE_ROLE) — por isso NÃO damos anon aqui.

----------------------------------------------------------------------
-- 1) contacts
----------------------------------------------------------------------
GRANT ALL ON public.contacts TO service_role;
GRANT SELECT ON public.contacts TO authenticated;

ALTER TABLE public.contacts ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS company_isolation_contacts ON public.contacts;
CREATE POLICY company_isolation_contacts
  ON public.contacts
  AS RESTRICTIVE
  FOR ALL
  TO authenticated
  USING (company_id = current_company_id())
  WITH CHECK (company_id = current_company_id());

DROP POLICY IF EXISTS contacts_select_tenant ON public.contacts;
CREATE POLICY contacts_select_tenant
  ON public.contacts
  FOR SELECT
  TO authenticated
  USING (company_id = current_company_id());

DROP POLICY IF EXISTS contacts_write_privileged ON public.contacts;
CREATE POLICY contacts_write_privileged
  ON public.contacts
  FOR ALL
  TO authenticated
  USING (has_role(auth.uid(), 'admin'::app_role) OR has_role(auth.uid(), 'manager'::app_role))
  WITH CHECK (has_role(auth.uid(), 'admin'::app_role) OR has_role(auth.uid(), 'manager'::app_role));

----------------------------------------------------------------------
-- 5) Storage bucket crm-meta-creatives — isolamento por path/company
----------------------------------------------------------------------
-- Limpa policies permissive antigas que só checavam bucket_id.
DROP POLICY IF EXISTS "Authenticated can view crm creatives" ON storage.objects;
DROP POLICY IF EXISTS authenticated_select_creatives ON storage.objects;
DROP POLICY IF EXISTS authenticated_update_creatives ON storage.objects;
DROP POLICY IF EXISTS authenticated_delete_creatives ON storage.objects;
DROP POLICY IF EXISTS public_select_creatives ON storage.objects;

-- Recria com isolamento por path. storage_path_belongs_to_current_company(name)
-- valida que o primeiro segmento do path corresponde ao slug/id da empresa ativa.
CREATE POLICY crm_creatives_select_tenant
  ON storage.objects
  FOR SELECT
  TO authenticated
  USING (
    bucket_id = 'crm-meta-creatives'
    AND storage_path_belongs_to_current_company(name)
  );

CREATE POLICY crm_creatives_insert_tenant
  ON storage.objects
  FOR INSERT
  TO authenticated
  WITH CHECK (
    bucket_id = 'crm-meta-creatives'
    AND storage_path_belongs_to_current_company(name)
    AND (has_role(auth.uid(), 'admin'::app_role)
      OR has_role(auth.uid(), 'manager'::app_role)
      OR has_role(auth.uid(), 'editor'::app_role))
  );

CREATE POLICY crm_creatives_update_tenant
  ON storage.objects
  FOR UPDATE
  TO authenticated
  USING (
    bucket_id = 'crm-meta-creatives'
    AND storage_path_belongs_to_current_company(name)
    AND (has_role(auth.uid(), 'admin'::app_role)
      OR has_role(auth.uid(), 'manager'::app_role)
      OR has_role(auth.uid(), 'editor'::app_role))
  );

CREATE POLICY crm_creatives_delete_tenant
  ON storage.objects
  FOR DELETE
  TO authenticated
  USING (
    bucket_id = 'crm-meta-creatives'
    AND storage_path_belongs_to_current_company(name)
    AND (has_role(auth.uid(), 'admin'::app_role)
      OR has_role(auth.uid(), 'manager'::app_role))
  );

NOTIFY pgrst, 'reload schema';

-- Verificação final
----------------------------------------------------------------------
SELECT c.relname,
       c.relrowsecurity AS rls_enabled,
       (SELECT count(*) FROM pg_policy p WHERE p.polrelid = c.oid) AS n_policies
FROM pg_class c
JOIN pg_namespace n ON n.oid = c.relnamespace
WHERE n.nspname='public'
  AND c.relname IN ('contacts','leads','event_faqs','event_lineups')
ORDER BY c.relname;

SELECT polname, polcmd
FROM pg_policy
WHERE polrelid='storage.objects'::regclass
  AND polname LIKE 'crm_creatives_%'
ORDER BY polname;
