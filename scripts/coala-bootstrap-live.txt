-- VALIDAÇÃO PÓS-BOOTSTRAP
select
  count(*) as total,
  count(forecast_id) as vinculados,
  count(*) filter (where needs_manual_link) as needs_manual_link,
  count(*) filter (where bootstrap_source = 'exact') as exact_,
  count(*) filter (where bootstrap_source = 'fuzzy') as fuzzy_,
  count(*) filter (where bootstrap_source = 'value_anchor') as value_anchor,
  count(*) filter (where bootstrap_source like 'ambiguous%') as ambiguous,
  count(*) filter (where bootstrap_source = 'no_match') as no_match
from public.coala_sync_row_state
where config_id = 'c4d00d98-3c71-455a-a5fc-244b13b58f65';
