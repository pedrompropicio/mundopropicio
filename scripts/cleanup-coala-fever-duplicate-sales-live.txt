-- ============================================================
-- LIMPEZA: apagar TODAS as vendas Fever do evento Coala Festival
-- Portugal 2026 antes de re-importar (a importacao anterior ficou
-- por cima da anterior, duplicando receita).
--
-- Executar em Cloud View > Run SQL com LIVE seleccionado.
-- Depois disto, voltar a fazer a importacao Fever na app.
-- ============================================================

DO $$
DECLARE
  v_event uuid;
  v_fever uuid;
  v_deleted int;
  v_lots_deleted int;
BEGIN
  SELECT id INTO v_event
  FROM events
  WHERE name ILIKE 'Coala Festival Portugal 2026%'
  ORDER BY date DESC
  LIMIT 1;

  IF v_event IS NULL THEN
    RAISE EXCEPTION 'Evento Coala 2026 nao encontrado';
  END IF;

  SELECT id INTO v_fever
  FROM financial_accounts
  WHERE name ILIKE '%fever%'
  LIMIT 1;

  IF v_fever IS NULL THEN
    RAISE EXCEPTION 'Conta Fever nao encontrada';
  END IF;

  WITH latest_lot_window AS (
    SELECT max(date_trunc('minute', l.created_at)) AS keep_minute
    FROM event_ticket_lots l
    JOIN event_ticket_zones z ON z.id = l.zone_id
    WHERE z.event_id = v_event
  ), old_lots AS (
    SELECT l.id
    FROM event_ticket_lots l
    JOIN event_ticket_zones z ON z.id = l.zone_id
    CROSS JOIN latest_lot_window w
    WHERE z.event_id = v_event
      AND date_trunc('minute', l.created_at) < w.keep_minute
  ), del_lot_sales AS (
    DELETE FROM ticket_sales s
    USING old_lots ol
    WHERE s.lot_id = ol.id
    RETURNING s.id
  ), del_lots AS (
    DELETE FROM event_ticket_lots l
    USING old_lots ol
    WHERE l.id = ol.id
    RETURNING l.id
  )
  SELECT count(*) INTO v_lots_deleted FROM del_lots;

  WITH del AS (
    DELETE FROM ticket_sales s
    USING event_ticket_zones z
    WHERE s.zone_id = z.id
      AND z.event_id = v_event
      AND s.financial_account_id = v_fever
    RETURNING s.id
  )
  SELECT count(*) INTO v_deleted FROM del;

  RAISE NOTICE 'Lotes antigos apagados: %. Vendas Fever apagadas: % (evento %, conta %)', v_lots_deleted, v_deleted, v_event, v_fever;
END $$;

-- Validacao: deve devolver 0 linhas
SELECT count(*) AS vendas_fever_restantes
FROM ticket_sales s
JOIN event_ticket_zones z ON z.id = s.zone_id
JOIN events e ON e.id = z.event_id
JOIN financial_accounts fa ON fa.id = s.financial_account_id
WHERE e.name ILIKE 'Coala Festival Portugal 2026%'
  AND fa.name ILIKE '%fever%';
