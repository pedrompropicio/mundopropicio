BEGIN;

    paid_amount=0,
    payment_date=NULL,

WHERE id IN (

-- 2) Voltar a nota a 'approved' e desvincular TX-mãe
UPDATE public.reimbursement_notes
SET status='approved',
    payment_transaction_id=NULL,
    paid_at=NULL,
    updated_at=now()
WHERE id='8f2e4926-fc7e-4bed-8295-531bb2062e34';

-- 3) Eliminar a transação-mãe gerada (€124,89)
DELETE FROM public.transactions
WHERE id='2d3cdef6-f7fc-4b18-86b9-18ae98a2ff0c';

-- 4) Verificação
SELECT 'NOTA' AS tipo, code, status, payment_transaction_id::text AS payment_tx
FROM public.reimbursement_notes WHERE code='R-006/2026'
UNION ALL
SELECT 'ITEM', t.id::text, t.status, t.paid_amount::text
FROM public.transactions t
JOIN public.reimbursement_note_items i ON i.transaction_id=t.id
WHERE i.reimbursement_note_id='8f2e4926-fc7e-4bed-8295-531bb2062e34'
UNION ALL
SELECT 'TX-MAE', '2d3cdef6-f7fc-4b18-86b9-18ae98a2ff0c',
       CASE WHEN EXISTS(SELECT 1 FROM public.transactions WHERE id='2d3cdef6-f7fc-4b18-86b9-18ae98a2ff0c')
            THEN 'AINDA EXISTE' ELSE 'APAGADA' END, NULL;

COMMIT;
