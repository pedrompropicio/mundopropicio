--   2) Run
--   3) Confirma a linha final por empresa:

--   5) Se qualquer linha falhar, corre: ROLLBACK;

BEGIN;

-- 1) Empresas afetadas: só as que ainda têm códigos 10.7.* de 4 níveis
CREATE TEMP TABLE _affected_companies AS
SELECT DISTINCT company_id
FROM public.account_categories
WHERE code ~ '^10\.7\.[0-9]{2}\.[0-9]{2}$';

-- 2) Criar pais L2 10.8 / 10.9 por empresa afetada, se ainda não existirem
INSERT INTO public.account_categories (id, code, name, type, parent_id, company_id, is_active, event_required)
SELECT gen_random_uuid(), v.code, v.name, 'expense', root10.id, root10.company_id, true, true
FROM public.account_categories root10
JOIN _affected_companies ac ON ac.company_id = root10.company_id
CROSS JOIN (VALUES ('10.8', 'Estrutura'), ('10.9', 'Tecnologia')) AS v(code, name)
WHERE root10.code = '10'
  AND NOT EXISTS (
    SELECT 1
    FROM public.account_categories existing
    WHERE existing.company_id = root10.company_id
      AND existing.code = v.code
  );

-- 3) Garantir 10.7 como pai L2 "Serviços"
UPDATE public.account_categories c
SET name = 'Serviços',
    parent_id = root10.id,
    is_active = true,
    event_required = true,
    updated_at = now()
FROM public.account_categories root10
JOIN _affected_companies ac ON ac.company_id = root10.company_id
WHERE c.company_id = root10.company_id
  AND c.code = '10.7'
  AND root10.code = '10';

-- 4) Se alguma BP/TX estiver ligada aos nós intermédios antigos, mover para a 1.ª folha do grupo
WITH map AS (
  SELECT g.id AS old_id, leaf.id AS new_id
  FROM public.account_categories g
  JOIN public.account_categories leaf
    ON leaf.company_id = g.company_id
   AND leaf.code = CASE g.code
     WHEN '10.7.01' THEN '10.7.01.01'
     WHEN '10.7.02' THEN '10.7.02.01'
     WHEN '10.7.03' THEN '10.7.03.01'
   END
  WHERE g.code IN ('10.7.01', '10.7.02', '10.7.03')
)
UPDATE public.transactions t
SET category_id = map.new_id
FROM map
WHERE t.category_id = map.old_id;

WITH map AS (
  SELECT g.id AS old_id, leaf.id AS new_id
  FROM public.account_categories g
  JOIN public.account_categories leaf
    ON leaf.company_id = g.company_id
   AND leaf.code = CASE g.code
     WHEN '10.7.01' THEN '10.7.01.01'
     WHEN '10.7.02' THEN '10.7.02.01'
     WHEN '10.7.03' THEN '10.7.03.01'
   END
  WHERE g.code IN ('10.7.01', '10.7.02', '10.7.03')
)
UPDATE public.event_forecasts f
SET category_id = map.new_id
FROM map
WHERE f.category_id = map.old_id;

-- 5) Libertar os códigos 10.7.01 / 10.7.02 / 10.7.03 para as folhas reais
UPDATE public.account_categories
SET code = '__OLD_' || code,
    is_active = false,
    updated_at = now()
WHERE code IN ('10.7.01', '10.7.02', '10.7.03')
  AND company_id IN (SELECT company_id FROM _affected_companies);

-- 6) Promover folhas L4 para L3, preservando os UUIDs originais
WITH map(old_code, new_code, parent_code) AS (
  VALUES
    ('10.7.01.01', '10.7.01', '10.7'),
    ('10.7.01.02', '10.7.02', '10.7'),
    ('10.7.01.03', '10.7.03', '10.7'),
    ('10.7.02.01', '10.8.01', '10.8'),
    ('10.7.02.02', '10.8.02', '10.8'),
    ('10.7.02.03', '10.8.03', '10.8'),
    ('10.7.03.01', '10.9.01', '10.9'),
    ('10.7.03.02', '10.9.02', '10.9'),
    ('10.7.03.03', '10.9.03', '10.9')
)
UPDATE public.account_categories leaf
SET code = map.new_code,
    parent_id = parent.id,
    is_active = true,
    event_required = true,
    updated_at = now()
FROM map, public.account_categories parent
WHERE leaf.code = map.old_code
  AND leaf.company_id IN (SELECT company_id FROM _affected_companies)
  AND parent.company_id = leaf.company_id
  AND parent.code = map.parent_code;

-- 7) Remover os nós intermédios antigos, agora sem filhos nem referências BP/TX
DELETE FROM public.account_categories
WHERE code IN ('__OLD_10.7.01', '__OLD_10.7.02', '__OLD_10.7.03')
  AND company_id IN (SELECT company_id FROM _affected_companies);

-- 8) Corrigir eventuais remanescentes com código antigo, preservando UUIDs
WITH map(old_code, new_code, parent_code) AS (
  VALUES
    ('10.7.01.01', '10.7.01', '10.7'),
    ('10.7.01.02', '10.7.02', '10.7'),
    ('10.7.01.03', '10.7.03', '10.7'),
    ('10.7.02.01', '10.8.01', '10.8'),
    ('10.7.02.02', '10.8.02', '10.8'),
    ('10.7.02.03', '10.8.03', '10.8'),
    ('10.7.03.01', '10.9.01', '10.9'),
    ('10.7.03.02', '10.9.02', '10.9'),
    ('10.7.03.03', '10.9.03', '10.9')
)
UPDATE public.account_categories leaf
SET code = map.new_code,
    parent_id = parent.id,
    is_active = true,
    event_required = true,
    updated_at = now()
FROM map, public.account_categories parent
WHERE leaf.code = map.old_code
  AND leaf.company_id IN (SELECT company_id FROM _affected_companies)
  AND parent.company_id = leaf.company_id
  AND parent.code = map.parent_code;

-- 9) Auditoria final real por parent_id + por código
WITH RECURSIVE tree AS (
  SELECT id, code, name, parent_id, company_id, 1 AS depth
  FROM public.account_categories
  WHERE parent_id IS NULL
  UNION ALL
  SELECT c.id, c.code, c.name, c.parent_id, c.company_id, t.depth + 1
  FROM public.account_categories c
  JOIN tree t ON c.parent_id = t.id
)
SELECT
  (SELECT count(*) FROM tree WHERE code LIKE '10.%' AND depth >= 4) AS l4_hierarquia,
  (SELECT count(*) FROM public.account_categories WHERE code ~ '^10\.[0-9]+\.[0-9]+\.[0-9]+$') AS l4_codigo,
  (SELECT count(*) FROM public.account_categories WHERE code ~ '^10\.[7-9]\.[0-9]{2}$') AS folhas_l3_10_7_9,
  (SELECT count(*) FROM public.account_categories WHERE code IN ('10.7', '10.8', '10.9')) AS pais_l2_10_7_9,
  (SELECT count(*) FROM public.event_forecasts f JOIN public.account_categories c ON c.id = f.category_id WHERE c.code ~ '^10\.[7-9]') AS bp_preservados,
  (SELECT count(*) FROM public.transactions t JOIN public.account_categories c ON c.id = t.category_id WHERE c.code ~ '^10\.[7-9]') AS tx_preservados,
  (SELECT count(*) FROM public.account_categories WHERE code LIKE '__OLD\_%' ESCAPE '\') AS antigos_grupos;
