-- ===========================================================
-- SEED COALA FESTIVAL PORTUGAL a partir de MUNDO PROPÍCIO
-- Ambiente: LIVE (executar no Supabase Dashboard > SQL Editor)
-- Copia: account_categories (161) + suppliers (93)
-- ===========================================================
-- IDs:
--   MP    = 7c858982-6ccd-47ca-bd65-e0dd3eebf01c
--   Coala = 7d831e59-6e82-427b-95a0-64904aae5dd2
-- Pré-condições:
--   - Coala tem 0 contas e 0 fornecedores (base zerada)
--   - MP tem 161 contas (3 níveis) e 93 fornecedores
-- Executar como role 'postgres' (default no SQL Editor).
-- ===========================================================
-- NOTA: A constraint global UNIQUE(code) em account_categories
-- será trocada por UNIQUE(company_id, code) — necessário para
-- multi-tenant. Em Test já está aplicado por migration; aqui
-- aplicamos manualmente em Live antes da cópia.
-- ===========================================================

BEGIN;

-- 0a) Garantir constraint multi-tenant em account_categories
ALTER TABLE public.account_categories
  DROP CONSTRAINT IF EXISTS account_categories_code_key;

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conrelid = 'public.account_categories'::regclass
      AND conname = 'account_categories_company_code_key'
  ) THEN
    ALTER TABLE public.account_categories
      ADD CONSTRAINT account_categories_company_code_key
      UNIQUE (company_id, code);
  END IF;
END $$;

-- 0b) Garantir índice multi-tenant em suppliers (nome único por empresa)
DROP INDEX IF EXISTS public.suppliers_name_unique;

CREATE UNIQUE INDEX IF NOT EXISTS suppliers_company_name_unique
  ON public.suppliers (company_id, lower(trim(name)))
  WHERE is_active = true;

-- 1) Plano de Contas com mapa antigo->novo (preserva parent_id)
WITH id_map AS (
  SELECT id AS old_id, gen_random_uuid() AS new_id
  FROM public.account_categories
  WHERE company_id = '7c858982-6ccd-47ca-bd65-e0dd3eebf01c'
),
src AS (
  SELECT ac.*, m.new_id
  FROM public.account_categories ac
  JOIN id_map m ON m.old_id = ac.id
  WHERE ac.company_id = '7c858982-6ccd-47ca-bd65-e0dd3eebf01c'
)
INSERT INTO public.account_categories
  (id, code, name, type, parent_id, is_active, event_required,
   company_id, created_at, updated_at)
SELECT
  s.new_id,
  s.code,
  s.name,
  s.type,
  pm.new_id,
  s.is_active,
  s.event_required,
  '7d831e59-6e82-427b-95a0-64904aae5dd2'::uuid,
  now(),
  now()
FROM src s
LEFT JOIN id_map pm ON pm.old_id = s.parent_id;

-- 2) Fornecedores
INSERT INTO public.suppliers
  (id, name, trade_name, nif, contact_name, email, phone, address,
   iban, iban_2, iban_3, swift_bic, swift_bic_2, swift_bic_3,
   payment_terms, category, notes, is_active, is_partner,
   company_id, created_at, updated_at)
SELECT
  gen_random_uuid(),
  name, trade_name, nif, contact_name, email, phone, address,
  iban, iban_2, iban_3, swift_bic, swift_bic_2, swift_bic_3,
  payment_terms, category, notes, is_active, is_partner,
  '7d831e59-6e82-427b-95a0-64904aae5dd2'::uuid,
  now(), now()
FROM public.suppliers
WHERE company_id = '7c858982-6ccd-47ca-bd65-e0dd3eebf01c';

-- 3) Verificação
SELECT
  (SELECT COUNT(*) FROM public.account_categories WHERE company_id='7d831e59-6e82-427b-95a0-64904aae5dd2') AS coala_categories,
  (SELECT COUNT(*) FROM public.account_categories WHERE company_id='7d831e59-6e82-427b-95a0-64904aae5dd2' AND parent_id IS NULL) AS coala_l1,
  (SELECT COUNT(*) FROM public.suppliers          WHERE company_id='7d831e59-6e82-427b-95a0-64904aae5dd2') AS coala_suppliers,
  (SELECT COUNT(*) FROM public.account_categories ac
     WHERE ac.company_id='7d831e59-6e82-427b-95a0-64904aae5dd2'
       AND ac.parent_id IS NOT NULL
       AND NOT EXISTS (SELECT 1 FROM public.account_categories p
                        WHERE p.id=ac.parent_id AND p.company_id=ac.company_id)
  ) AS orphan_parents;
-- Esperado: 161, 10–11, 93, 0

COMMIT;
