-- Remove agendamento anterior (se existir) — ignora erro se ainda não existe
DO $$ BEGIN
  PERFORM cron.unschedule('coala-sync-auto');
EXCEPTION WHEN OTHERS THEN
  RAISE NOTICE 'coala-sync-auto não estava agendado, continuar.';
END $$;

-- Agenda novo: a cada 15 minutos, dispara dry_run em todas as configs ativas.
-- A escalada para auto_apply é feita dentro da edge function quando reviewCount=0.
SELECT cron.schedule(
  'coala-sync-auto',
  '*/15 * * * *',
  $cron$
  SELECT net.http_post(
    url := 'https://sfohvvlqccmmebvjgibx.supabase.co/functions/v1/sync-coala-from-drive',
    headers := jsonb_build_object(
      'Content-Type', 'application/json',
      'Authorization', 'Bearer ' || (
        SELECT decrypted_secret FROM vault.decrypted_secrets
        WHERE name = 'email_queue_service_role_key'
      )
    ),
    body := jsonb_build_object(
      'mode','dry_run',
      'configId','c4d00d98-3c71-455a-a5fc-244b13b58f65',
      'triggeredBy','cron'
    )
  );
  $cron$
);

-- Verificação rápida
SELECT jobid, schedule, jobname, active FROM cron.job WHERE jobname = 'coala-sync-auto';
