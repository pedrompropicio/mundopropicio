SELECT ac.code, ac.name, ac.is_active, (SELECT COUNT(*) FROM event_forecasts WHERE category_id = ac.id) AS bp_lines, (SELECT COALESCE(SUM(amount),0) FROM event_forecasts WHERE category_id = ac.id) AS bp_total, (SELECT COUNT(*) FROM transactions WHERE category_id = ac.id) AS tx_count, (SELECT COALESCE(SUM(amount),0) FROM transactions WHERE category_id = ac.id) AS tx_total, (SELECT COUNT(*) FROM event_cache_payments WHERE category_id = ac.id) AS cache_pay, (SELECT COUNT(*) FROM event_closing_costs WHERE category_id = ac.id) AS fechos, (SELECT COUNT(*) FROM recurring_transactions WHERE category_id = ac.id) AS recorrentes, (SELECT COUNT(*) FROM camarim_items WHERE category_id = ac.id) AS camarim, (SELECT COUNT(*) FROM event_cache_deductions WHERE category_id = ac.id) AS cache_ded, (SELECT COUNT(*) FROM quotations WHERE category_id = ac.id) AS quotations FROM public.account_categories ac WHERE ac.code LIKE '4.2.%' ORDER BY ac.code;
