-- Reatribuir patrocínios do Coala à categoria 1.2.01 da empresa Coala
-- (estavam a apontar para o ID da categoria 1.2.01 da Mundo Propício, daí "Z Sem categoria")
-- Executar em Cloud View > SQL > LIVE

BEGIN;

UPDATE public.event_forecasts ef
SET category_id = '3312f0f0-b5ae-4a74-94b5-32fda4c8b920'
FROM public.events e
WHERE ef.event_id = e.id
  AND e.company_id = '7d831e59-6e82-427b-95a0-64904aae5dd2'
  AND ef.category_id = 'c0000004-0000-0000-0000-000000000001';

UPDATE public.transactions t
SET category_id = '3312f0f0-b5ae-4a74-94b5-32fda4c8b920'
FROM public.events e
WHERE t.event_id = e.id
  AND e.company_id = '7d831e59-6e82-427b-95a0-64904aae5dd2'
  AND t.category_id = 'c0000004-0000-0000-0000-000000000001';

-- Validação (esperado: 0 linhas)
SELECT 'Forecasts ainda com category errado:' AS info, COUNT(*) AS qtd
FROM public.event_forecasts ef
JOIN public.events e ON e.id = ef.event_id
WHERE e.company_id = '7d831e59-6e82-427b-95a0-64904aae5dd2'
  AND ef.category_id = 'c0000004-0000-0000-0000-000000000001'
UNION ALL
SELECT 'Transactions ainda com category errado:', COUNT(*)
FROM public.transactions t
JOIN public.events e ON e.id = t.event_id
WHERE e.company_id = '7d831e59-6e82-427b-95a0-64904aae5dd2'
  AND t.category_id = 'c0000004-0000-0000-0000-000000000001';

COMMIT;
