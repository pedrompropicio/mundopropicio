-- ─────────────────────────────────────────────────────────────────────────────
-- PASSO 1 — Estado actual: master + splits + campanhas Simone activas
-- ─────────────────────────────────────────────────────────────────────────────

-- 1a. Eventos Simone (master + splits)
SELECT id, name, event_type, parent_event_id, date, status
FROM public.events
WHERE name ILIKE '%simone%mendes%'
ORDER BY event_type DESC NULLS LAST, date NULLS LAST;

-- 1b. Campanhas Simone activas + linkage actual
SELECT
  c.id              AS campaign_uuid,
  c.external_campaign_id,
  c.name            AS campaign_name,
  c.linked_event_id,
  c.linked_event_locked,
  e.name            AS linked_event_name,
  e.event_type      AS linked_event_type,
  e.parent_event_id AS linked_event_parent_id
FROM crm.meta_campaign_snapshot c
LEFT JOIN public.events e ON e.id = c.linked_event_id
WHERE c.name ILIKE '%simone%'
  AND c.status = 'ACTIVE'
ORDER BY c.name;

-- ─────────────────────────────────────────────────────────────────────────────
-- PASSO 2 — Preview do re-mapping proposto (não escreve nada)
-- ─────────────────────────────────────────────────────────────────────────────
-- Heurística: se nome da campanha contém "porto" → split Porto;
--             se contém "lisboa" → split Lisboa;
--             caso contrário → mantém o linkage actual (provavelmente master).
-- O preview mostra side-by-side current vs proposed para tu confirmares.

WITH simone_master AS (
  SELECT id, name FROM public.events
  WHERE name ILIKE '%simone%mendes%'
    AND event_type = 'tour_master'
  LIMIT 1
),
simone_splits AS (
  SELECT s.id, s.name, s.parent_event_id,
         CASE
           WHEN unaccent(lower(s.name)) ILIKE '%porto%'  THEN 'porto'
           WHEN unaccent(lower(s.name)) ILIKE '%lisboa%' THEN 'lisboa'
           ELSE NULL
         END AS city_tag
  FROM public.events s
  JOIN simone_master m ON s.parent_event_id = m.id
  WHERE s.event_type = 'tour_split'
),
campaign_proposal AS (
  SELECT
    c.id AS campaign_uuid,
    c.external_campaign_id,
    c.name AS campaign_name,
    c.linked_event_id AS current_event_id,
    e_curr.name AS current_event_name,
    CASE
      WHEN unaccent(lower(c.name)) ILIKE '%porto%'  THEN (SELECT id FROM simone_splits WHERE city_tag = 'porto'  LIMIT 1)
      WHEN unaccent(lower(c.name)) ILIKE '%lisboa%' THEN (SELECT id FROM simone_splits WHERE city_tag = 'lisboa' LIMIT 1)
      ELSE c.linked_event_id
    END AS proposed_event_id
  FROM crm.meta_campaign_snapshot c
  LEFT JOIN public.events e_curr ON e_curr.id = c.linked_event_id
  WHERE c.name ILIKE '%simone%'
    AND c.status = 'ACTIVE'
)
SELECT
  cp.campaign_uuid,
  cp.external_campaign_id,
  cp.campaign_name,
  cp.current_event_name,
  e_prop.name AS proposed_event_name,
  e_prop.event_type AS proposed_event_type,
  CASE
    WHEN cp.proposed_event_id = cp.current_event_id THEN 'no-op'
    WHEN cp.proposed_event_id IS NULL              THEN 'skip (ambíguo)'
    ELSE 're-link'
  END AS action
FROM campaign_proposal cp
LEFT JOIN public.events e_prop ON e_prop.id = cp.proposed_event_id
ORDER BY cp.campaign_name;
